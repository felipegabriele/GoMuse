# Pulse and HRV Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Show live heart rate (bpm) and RMSSD heart-rate variability in the gomuse viewer, derived from whichever pulse stream the headband produces — the Athena's 16-channel `Optics` or a classic Muse's 3-channel `PPG`.

**Architecture:** A new `internal/pulse` package holds four small units in a chain — `Selector` picks the best channel, `Detector` finds beats, `HRV` turns beats into numbers, `Tracker` wires them together and exposes a `Snapshot`. The server holds one `Tracker` per WebSocket client (like the existing `dsp` banks) and pushes a JSON `pulse` message at ~4 Hz. `app.js` renders a new Pulse panel and computes nothing.

**Tech Stack:** Go 1.x standard library only, `internal/dsp` for the DC blocker, uPlot (already vendored) for the chart. No new dependencies.

Spec: `docs/superpowers/specs/2026-07-27-hrv-design.md`

```go
package pulse

import "testing"

// ramp builds a channel of n samples: a constant pedestal plus a triangle wave
// of the given fractional modulation. A triangle is used rather than a sine
// because its p5 and p95 are easy to reason about.
func ramp(n int, pedestal, modulation float64) []float64 {
	out := make([]float64, n)
	for i := range out {
		phase := float64(i%20) / 20.0 // 20-sample period
		if phase > 0.5 {
			phase = 1 - phase
		}
		out[i] = pedestal * (1 + modulation*(phase*4-1))
	}
	return out
}
```

```go
// Package pulse turns a headband's raw pulse stream into a heart rate and an
// HRV figure.
//
// It is deliberately independent of package sample: it takes channel-major
// [][]float64 the way internal/dsp does, so it can be unit-tested on synthetic
// waveforms with no notion of a device or a stream.
//
// The chain is Selector -> Detector -> HRV, assembled by Tracker.
package pulse

import "sort"

const (
	// WindowSec is how much history the channel ranking looks at, and EvalSec
	// how often it re-ranks. Four seconds spans three beats at a resting rate,
	// enough for the percentile spread to reflect the pulse rather than a
	// single beat's shape.
	WindowSec = 4.0
	EvalSec   = 2.0

	// ModulationFloor is the smallest (p95-p5)/mean that counts as a pulse.
	// Measured on a MuseS-1FF3 at preset p1041: OPT12/13 reach ~0.04 and
	// OPT8/9 ~0.02, while OPT4-7 sit at ~0.0006. This floor admits the former
	// and excludes the latter.
	ModulationFloor = 0.001

	// ChallengerMargin is how much better a channel must be to unseat the
	// incumbent. Without it, two near-equal channels swap every evaluation and
	// the displayed channel name flickers.
	ChallengerMargin = 1.20

	// MinMean is the absolute floor below which a channel is treated as
	// carrying nothing. decodeAthenaOptics zero-fills the canonical channels a
	// tag does not carry (tag 0x34 fills only slots 4-7, 0x35 only 0-7), so on
	// presets other than p1041 some channels are identically zero and their
	// mean is exactly 0. Dividing by that is 0/0. Go yields NaN, which happens
	// to lose every > comparison and so would never be selected, but relying on
	// that is accidental. One raw count is far below any real optical reading.
	MinMean = 1.0
)

// Selector ranks pulse channels by how strongly they modulate and picks one.
//
// Which channel carries the best pulse depends on headband fit and on the
// preset, so a fixed choice is wrong: on the reference capture the best channel
// modulates sixty times more than the worst.
type Selector struct {
	window int // samples retained per channel
	every  int // samples between evaluations

	buf   [][]float64 // rolling history, channel-major
	since int         // samples seen since the last evaluation

	chosen int
	score  float64
}

// NewSelector returns a selector for a stream of the given rate and width.
func NewSelector(rate float64, channels int) *Selector {
	if rate <= 0 {
		panic("pulse: rate must be positive")
	}
	if channels <= 0 {
		panic("pulse: channels must be positive")
	}
	s := &Selector{
		window: int(rate * WindowSec),
		every:  int(rate * EvalSec),
		buf:    make([][]float64, channels),
		chosen: -1,
	}
	if s.window < 1 {
		s.window = 1
	}
	if s.every < 1 {
		s.every = 1
	}
	return s
}

// Observe appends a channel-major block and re-ranks when enough new samples
// have arrived. It does not retain or modify data.
func (s *Selector) Observe(data [][]float64) {
	if len(data) != len(s.buf) || len(data) == 0 {
		return
	}
	n := len(data[0])
	if n == 0 {
		return
	}

	for c := range s.buf {
		s.buf[c] = append(s.buf[c], data[c]...)
		if excess := len(s.buf[c]) - s.window; excess > 0 {
			s.buf[c] = append(s.buf[c][:0], s.buf[c][excess:]...)
		}
	}

	s.since += n
	if s.since >= s.every && len(s.buf[0]) >= s.window {
		s.since = 0
		s.evaluate()
	}
}

// Channel returns the selected channel index, or -1 when none qualifies.
func (s *Selector) Channel() int { return s.chosen }

func (s *Selector) evaluate() {
	best, bestScore := -1, 0.0
	for c := range s.buf {
		score := modulation(s.buf[c])
		if score > bestScore {
			best, bestScore = c, score
		}
	}

	if best < 0 || bestScore < ModulationFloor {
		s.chosen, s.score = -1, 0
		return
	}
	// Hold the incumbent unless the challenger clears the margin.
	if s.chosen >= 0 && best != s.chosen && bestScore < s.score*ChallengerMargin {
		// Keep the incumbent, but refresh its score so a decaying incumbent
		// can eventually be beaten.
		s.score = modulation(s.buf[s.chosen])
		if s.score >= ModulationFloor {
			return
		}
	}
	s.chosen, s.score = best, bestScore
}

// modulation is (p95-p5)/mean, a scale-free measure of how much a channel
// swings relative to its own pedestal. Returns 0 for a channel carrying
// nothing, so it can never be selected.
func modulation(x []float64) float64 {
	if len(x) < 2 {
		return 0
	}
	sum := 0.0
	for _, v := range x {
		sum += v
	}
	mean := sum / float64(len(x))
	if mean <= MinMean {
		return 0
	}

	sorted := make([]float64, len(x))
	copy(sorted, x)
	sort.Float64s(sorted)
	return (percentile(sorted, 0.95) - percentile(sorted, 0.05)) / mean
}

// percentile indexes into an already-sorted slice by nearest rank.
func percentile(sorted []float64, p float64) float64 {
	i := int(p * float64(len(sorted)-1))
	if i < 0 {
		i = 0
	}
	if i >= len(sorted) {
		i = len(sorted) - 1
	}
	return sorted[i]
}
```

```go
package pulse

import (

// pulseTrain builds a PPG-like waveform: a large DC pedestal with a periodic
// asymmetric bump, matching the shape internal/synth generates.
func pulseTrain(n int, rate, bpm, pedestal, modulation float64) (samples, ts []float64) {
	samples = make([]float64, n)
	ts = make([]float64, n)
	hz := bpm / 60
	for i := range samples {
		e := float64(i) / rate
		phase := math.Mod(e*hz, 1)
		// A sharp systolic rise and slower decay, peaking early in the cycle.
		bump := math.Exp(-math.Pow((phase-0.2)/0.12, 2))
		samples[i] = pedestal * (1 + modulation*bump)
		ts[i] = 1000 + e
	}
	return samples, ts
}

func TestDetectorFindsKnownRate(t *testing.T) {
	const rate, bpm = 64.0, 72.0
	samples, ts := pulseTrain(64*20, rate, bpm, 100000, 0.04) // 20 s

	d := NewDetector(rate)
	beats, gap := d.Process(samples, ts)
	if gap {
		t.Error("gap reported on a continuous stream")
	}

	// 20 s at 72 bpm is 24 beats. Allow the first to be missed while the peak
	// estimate settles.
	if len(beats) < 22 || len(beats) > 25 {
		t.Fatalf("detected %d beats over 20 s at 72 bpm, want 22-25", len(beats))
	}

	// Mean interval should land within 2 bpm of the truth.
	span := beats[len(beats)-1] - beats[0]
	got := 60 * float64(len(beats)-1) / span
	if math.Abs(got-bpm) > 2 {
		t.Errorf("implied rate = %.1f bpm, want %.0f +/- 2", got, bpm)
	}
}

func TestDetectorRefractoryRejectsDoublePeaks(t *testing.T) {
	const rate = 64.0
	// A dicrotic notch: a second, smaller bump 100 ms after each beat, well
	// inside the refractory period. Real PPG has one; counting it doubles the
	// reported rate.
	n := int(rate * 10)
	samples := make([]float64, n)
	ts := make([]float64, n)
	for i := range samples {
		e := float64(i) / rate
		phase := math.Mod(e*1.2, 1) // 72 bpm
		main := math.Exp(-math.Pow((phase-0.2)/0.10, 2))
		notch := 0.5 * math.Exp(-math.Pow((phase-0.32)/0.06, 2))
		samples[i] = 100000 * (1 + 0.04*(main+notch))
		ts[i] = 1000 + e
	}

	d := NewDetector(rate)
	beats, _ := d.Process(samples, ts)

	// 10 s at 72 bpm is 12 beats. Counting notches would give ~24.
	if len(beats) > 14 {
		t.Errorf("detected %d beats, want <=14: the notch is being counted", len(beats))
	}
}

func TestDetectorReportsGap(t *testing.T) {
	const rate = 64.0
	samples, ts := pulseTrain(int(rate*5), rate, 72, 100000, 0.04)

	d := NewDetector(rate)
	if _, gap := d.Process(samples, ts); gap {
		t.Error("gap reported on the first block")
	}

	// A second block starting 2 s after the first ended: samples were lost.
	next, nextTS := pulseTrain(int(rate*5), rate, 72, 100000, 0.04)
	last := ts[len(ts)-1]
	for i := range nextTS {
		nextTS[i] = last + 2.0 + float64(i)/rate
	}
	if _, gap := d.Process(next, nextTS); !gap {
		t.Error("no gap reported across a 2 s discontinuity")
	}
}

func TestDetectorContiguousBlocksReportNoGap(t *testing.T) {
	const rate = 64.0
	samples, ts := pulseTrain(int(rate*10), rate, 72, 100000, 0.04)

	d := NewDetector(rate)
	// Feed in 16-sample blocks, as real chunks arrive.
	for i := 0; i+16 <= len(samples); i += 16 {
		if _, gap := d.Process(samples[i:i+16], ts[i:i+16]); gap {
			t.Fatalf("gap reported at contiguous block starting %d", i)
		}
	}
}
```

```go
package pulse

import (
	"math"

	"gomuse/internal/dsp"
)

const (
	// RefractorySec is the shortest accepted gap between beats: 300 ms caps
	// detection at 200 bpm and, more usefully, rejects the dicrotic notch —
	// the secondary bump every real PPG waveform carries about 100 ms after
	// systole, which would otherwise double the reported rate.
	RefractorySec = 0.30

	// PeakFraction of the running peak estimate is the detection threshold.
	PeakFraction = 0.5

	// PeakTauSec is how fast the peak estimate decays toward the current
	// signal. Three seconds is a few beats: long enough not to be dragged down
	// between beats, short enough to follow changing perfusion.
	PeakTauSec = 3.0

	// SmoothTaps is the moving-average length, ~78 ms at 64 Hz. Enough to kill
	// sample-rate noise without blunting the systolic rise.
	SmoothTaps = 5

	// GapPeriods is how far apart two consecutive samples may be before the
	// stream is treated as having lost data. 1.5 periods tolerates ordinary
	// jitter in the dejittered timestamps while catching a single lost packet.
	GapPeriods = 1.5

	// dcCornerHz is the DC blocker corner. The pedestal is ~100000 counts and
	// the pulse ~4000, so removing DC is what makes a fixed threshold possible
	// at all.
	//
	// A dsp.FirwinBandpass is deliberately not used: at 64 Hz a 0.7 Hz lower
	// edge needs roughly 512 taps and ~4 s of group delay, unusable live. This
	// is the same limitation dcblock.go documents for EEG, and the same remedy.
	dcCornerHz = 0.5
)

// Detector finds beats in one pulse channel.
type Detector struct {
	rate   float64
	period float64

	dc *dsp.DCBlocker

	hist  []float64 // moving-average window
	sum   float64
	idx   int
	full  bool

	peak      float64
	peakDecay float64

	// A beat is the middle of three smoothed samples, so two must be kept.
	prev, prevPrev float64
	prevTS         float64
	have           int

	lastBeat float64
	lastTS   float64
	primed   bool
}

// NewDetector returns a detector for a stream of the given rate.
func NewDetector(rate float64) *Detector {
	if rate <= 0 {
		panic("pulse: rate must be positive")
	}
	d := &Detector{
		rate:      rate,
		period:    1 / rate,
		hist:      make([]float64, SmoothTaps),
		peakDecay: math.Exp(-1 / (rate * PeakTauSec)),
	}
	d.dc = dsp.NewDCBlocker(dcCornerHz, rate)
	return d
}

// Process feeds one channel's samples and their timestamps, returning the beat
// timestamps found within this call. gap reports that samples were lost between
// the previous call and this one, so the caller must not form an interval
// across the boundary.
//
// samples and timestamps must be the same length and are not modified.
func (d *Detector) Process(samples, timestamps []float64) (beats []float64, gap bool) {
	if len(samples) == 0 || len(samples) != len(timestamps) {
		return nil, false
	}

	if d.primed && timestamps[0]-d.lastTS > GapPeriods*d.period {
		// Data was lost. The filter and peak state describe a signal that no
		// longer continues, so start them over rather than carrying a stale
		// history across the discontinuity.
		gap = true
		d.resetSignalState()
	}

	filtered := d.dc.Process(samples)

	for i, v := range filtered {
		s := d.smooth(v)

		if s > d.peak {
			d.peak = s
		} else {
			d.peak *= d.peakDecay
		}

		if d.have >= 2 {
			threshold := PeakFraction * d.peak
			isPeak := d.prev > d.prevPrev && d.prev >= s && d.prev > threshold
			if isPeak && (!d.primed || d.prevTS-d.lastBeat >= RefractorySec) {
				beats = append(beats, d.prevTS)
				d.lastBeat = d.prevTS
				d.primed = true
			}
		}

		d.prevPrev, d.prev = d.prev, s
		d.prevTS = timestamps[i]
		if d.have < 2 {
			d.have++
		}
	}

	d.lastTS = timestamps[len(timestamps)-1]
	d.primed = true
	return beats, gap
}

// Reset clears all state, as when the selected channel changes.
func (d *Detector) Reset() {
	d.resetSignalState()
	d.lastBeat, d.lastTS, d.primed = 0, 0, false
}

// resetSignalState clears everything derived from sample values, leaving the
// beat history alone.
func (d *Detector) resetSignalState() {
	d.dc.Reset()
	for i := range d.hist {
		d.hist[i] = 0
	}
	d.sum, d.idx, d.full = 0, 0, false
	d.peak = 0
	d.prev, d.prevPrev, d.have = 0, 0, 0
}

// smooth is a moving average over SmoothTaps samples.
func (d *Detector) smooth(v float64) float64 {
	d.sum -= d.hist[d.idx]
	d.hist[d.idx] = v
	d.sum += v
	d.idx++
	if d.idx == len(d.hist) {
		d.idx, d.full = 0, true
	}
	n := len(d.hist)
	if !d.full {
		n = d.idx
	}
	if n == 0 {
		return 0
	}
	return d.sum / float64(n)
}
```

```go
package pulse

import (
	"math"
	"testing"
)

// addIntervals feeds beats spaced by the given millisecond intervals, starting
// at t=1000, and returns the timestamp of the final beat.
func addIntervals(h *HRV, ms []float64) float64 {
	t := 1000.0
	h.AddBeat(t)
	for _, d := range ms {
		t += d / 1000
		h.AddBeat(t)
	}
	return t
}

// The golden RMSSD, computed by hand from the interval sequence below:
//
//	intervals 800 840 810 850 820 860
//	diffs       40 -30  40 -30  40
//	none excluded: the largest ratio is 40/800 = 5%, under EctopicRatio
//	squares   1600 900 1600 900 1600  -> sum 6600, mean 1320
//	RMSSD = sqrt(1320) = 36.331804...
func TestRMSSDOfGoldenSequence(t *testing.T) {
	got, ok := rmssdOf([]float64{800, 840, 810, 850, 820, 860})
	if !ok {
		t.Fatal("rmssdOf reported not enough data for 6 intervals")
	}
	want := math.Sqrt(1320)
	if math.Abs(got-want) > 1e-9 {
		t.Errorf("rmssdOf = %.9f, want %.9f", got, want)
	}
}

func TestRMSSDExcludesEctopicDifferences(t *testing.T) {
	// A doubled interval in the middle. Both differences it takes part in
	// exceed EctopicRatio and must be excluded, leaving the surrounding steady
```

```go
package pulse

import (
	"math"
	"sort"
)

const (
	// MinIntervalMs and MaxIntervalMs bound a plausible beat-to-beat interval:
	// 200 bpm down to 30 bpm. Anything outside is a detection error, not a
	// heartbeat.
	MinIntervalMs = 300.0
	MaxIntervalMs = 2000.0

	// LockIntervals is how many clean intervals must accumulate before any
	// number is shown. Reporting a rate off one or two beats would be
	// confidently wrong.
	LockIntervals = 5

	// BPMWindow is how many recent intervals the displayed rate is taken over.
	BPMWindow = 7

	// RMSSDWindowSec is the analysis window, and RMSSDMinSpanSec how much must
	// accumulate before RMSSD is shown at all. RMSSD over a handful of seconds
	// is dominated by whichever beats happen to be in it.
	RMSSDWindowSec  = 60.0
	RMSSDMinSpanSec = 30.0

	// EctopicRatio is the fractional difference between successive intervals
	// above which the pair is treated as an artifact — a missed or extra beat —
	// and excluded from RMSSD. This biases RMSSD slightly low in exchange for
	// not letting one bad beat dominate a metric built from small differences.
	EctopicRatio = 0.20

	// BeatHistorySec is how long beat timestamps are kept for display. It
	// matches MAX_WINDOW in app.js, the viewer's longest selectable window.
	BeatHistorySec = 60.0
)

// interval is one accepted beat-to-beat gap, tagged with when it ended so the
// analysis window can be trimmed by time rather than by count.
type interval struct {
	at float64 // Unix seconds of the closing beat
	ms float64
}

// HRV turns a stream of beat timestamps into a rate and an RMSSD figure.
type HRV struct {
	last     float64
	haveLast bool

	ivals []interval
	beats []float64
}

// NewHRV returns an empty HRV accumulator.
func NewHRV() *HRV { return &HRV{} }

// AddBeat records a beat at time t, in Unix seconds.
func (h *HRV) AddBeat(t float64) {
	h.beats = append(h.beats, t)
	if cut := t - BeatHistorySec; len(h.beats) > 0 {
		i := 0
		for i < len(h.beats) && h.beats[i] < cut {
			i++
		}
		h.beats = append(h.beats[:0], h.beats[i:]...)
	}

	if h.haveLast {
		if ms := (t - h.last) * 1000; ms >= MinIntervalMs && ms <= MaxIntervalMs {
			h.ivals = append(h.ivals, interval{at: t, ms: ms})
			cut := t - RMSSDWindowSec
			i := 0
			for i < len(h.ivals) && h.ivals[i].at < cut {
				i++
			}
			h.ivals = append(h.ivals[:0], h.ivals[i:]...)
		}
	}
	h.last, h.haveLast = t, true
}

// Break marks a discontinuity: the next beat starts a new interval chain rather
// than closing one across lost data. See the Detector's gap return.
func (h *HRV) Break() { h.haveLast = false }

// IntervalCount returns how many accepted intervals are in the window.
func (h *HRV) IntervalCount() int { return len(h.ivals) }

// BPM returns the current rate, and whether enough intervals exist to report
// one. The median of the recent intervals is used rather than the mean so that
// a single missed or doubled beat does not move the displayed number.
func (h *HRV) BPM() (float64, bool) {
	if len(h.ivals) < LockIntervals {
		return 0, false
	}
	n := BPMWindow
	if n > len(h.ivals) {
		n = len(h.ivals)
	}
	recent := make([]float64, n)
	for i, v := range h.ivals[len(h.ivals)-n:] {
		recent[i] = v.ms
	}
	m := medianOf(recent)
	if m <= 0 {
		return 0, false
	}
	return 60000 / m, true
}

// LastIntervalMs returns the most recent accepted interval.
func (h *HRV) LastIntervalMs() (float64, bool) {
	if len(h.ivals) == 0 {
		return 0, false
	}
	return h.ivals[len(h.ivals)-1].ms, true
}

// RMSSD returns the root mean square of successive differences over the window,
// and whether enough time has accumulated to report it.
func (h *HRV) RMSSD() (float64, bool) {
	if len(h.ivals) < LockIntervals {
		return 0, false
	}
	span := h.ivals[len(h.ivals)-1].at - h.ivals[0].at
	if span < RMSSDMinSpanSec {
		return 0, false
	}
	ms := make([]float64, len(h.ivals))
	for i, v := range h.ivals {
		ms[i] = v.ms
	}
	return rmssdOf(ms)
}

// Beats returns the beat timestamps at or after since, oldest first.
func (h *HRV) Beats(since float64) []float64 {
	out := make([]float64, 0, len(h.beats))
	for _, t := range h.beats {
		if t >= since {
			out = append(out, t)
		}
	}
	return out
}

// Reset clears all accumulated state.
func (h *HRV) Reset() {
	h.last, h.haveLast = 0, false
	h.ivals = h.ivals[:0]
	h.beats = h.beats[:0]
}

// medianOf returns the median of ms. It sorts a copy, so the caller's slice is
// untouched. For an even count it averages the two middle values.
func medianOf(ms []float64) float64 {
	if len(ms) == 0 {
		return 0
	}
	s := make([]float64, len(ms))
	copy(s, ms)
	sort.Float64s(s)
	mid := len(s) / 2
	if len(s)%2 == 1 {
		return s[mid]
	}
	return (s[mid-1] + s[mid]) / 2
}

// rmssdOf is the root mean square of successive differences, excluding pairs
// whose fractional difference exceeds EctopicRatio. It reports false when too
// few differences survive to mean anything.
func rmssdOf(ms []float64) (float64, bool) {
	if len(ms) < 2 {
		return 0, false
	}
	sum, n := 0.0, 0
	for i := 1; i < len(ms); i++ {
		a, b := ms[i-1], ms[i]
		lo := math.Min(a, b)
		if lo <= 0 {
			continue
		}
		if math.Abs(a-b)/lo > EctopicRatio {
			continue // an artifact, not a beat-to-beat variation
		}
		d := a - b
		sum += d * d
		n++
	}
	if n < 2 {
		return 0, false
	}
	return math.Sqrt(sum / float64(n)), true
}
```

```go
package pulse

// State is what the tracker can honestly say about the pulse right now.
//
// Three states exist rather than two so the viewer never shows a fabricated
// number: "no pulse" and "acquiring" look identical to a user reading bpm, but
// they mean different things and only one of them will resolve by waiting.
type State string

const (
	// StateNoPulse means no channel carries enough modulation — the headband
	// is off the head, or the preset sends nothing usable.
	StateNoPulse State = "no pulse"
	// StateAcquiring means a channel is selected but too few clean intervals
	// have accumulated to report a rate.
	StateAcquiring State = "acquiring"
	// StateLocked means the numbers are real.
	StateLocked State = "locked"
)

// Snapshot is the tracker's state at one instant, safe to serialise.
//
// The Has* flags exist rather than sentinel values because "no number yet" and
// "a rate of zero" must not be confusable.
type Snapshot struct {
	State     State
	Channel   int // -1 when none is selected
	BPM       float64
	HasBPM    bool
	RMSSD     float64
	HasRMSSD  bool
	IBIMs     float64
	HasIBI    bool
	BeatCount int
	Beats     []float64
}

// Tracker wires Selector, Detector and HRV into one unit: it takes raw
// channel-major pulse samples and answers with a rate and an HRV figure.
type Tracker struct {
	sel *Selector
	det *Detector
	hrv *HRV

	channel int
	lastTS  float64
}

// NewTracker returns a tracker for a pulse stream of the given rate and width.
func NewTracker(rate float64, channels int) *Tracker {
	return &Tracker{
		sel:     NewSelector(rate, channels),
		det:     NewDetector(rate),
		hrv:     NewHRV(),
		channel: -1,
	}
}

// Process feeds one chunk of the pulse stream. data is channel-major and is
// neither retained nor modified.
func (t *Tracker) Process(data [][]float64, timestamps []float64) {
	if len(data) == 0 || len(timestamps) == 0 {
		return
	}
	t.lastTS = timestamps[len(timestamps)-1]

	t.sel.Observe(data)
	ch := t.sel.Channel()

	if ch != t.channel {
		// The new channel has a different pedestal and unrelated phase, so the
		// filter history and the pending interval both describe a signal that
		// is no longer being watched.
		t.det.Reset()
		t.hrv.Break()
		t.channel = ch
	}
	if ch < 0 || ch >= len(data) {
		return
	}

	beats, gap := t.det.Process(data[ch], timestamps)
	if gap {
		t.hrv.Break()
	}
	for _, b := range beats {
		t.hrv.AddBeat(b)
	}
}

// Snapshot reports the current state. Beats covers the last BeatHistorySec.
func (t *Tracker) Snapshot() Snapshot {
	s := Snapshot{Channel: t.channel, State: StateNoPulse}
	if t.channel < 0 {
		return s
	}

	s.State = StateAcquiring
	s.Beats = t.hrv.Beats(t.lastTS - BeatHistorySec)
	s.BeatCount = len(s.Beats)

	bpm, ok := t.hrv.BPM()
	if !ok {
		return s
	}
	s.State, s.BPM, s.HasBPM = StateLocked, bpm, true
	s.RMSSD, s.HasRMSSD = t.hrv.RMSSD()
	s.IBIMs, s.HasIBI = t.hrv.LastIntervalMs()
	return s
}
```

```go
		// be nil, and a zero-channel bank is valid.

// pulseUpdate feeds a chunk to the pulse tracker and reports whether a message
// is due. It returns false for chunks of any other stream, for devices with no
// pulse stream, and between throttle intervals.
func (c *client) pulseUpdate(chunk sample.Chunk) (PulseMessage, bool) {
	if c.tracker == nil || chunk.Stream != c.pulseID || chunk.NumSamples() == 0 {
		return PulseMessage{}, false
	}
	c.tracker.Process(chunk.Data, chunk.Timestamps)

	now := chunk.Timestamps[chunk.NumSamples()-1]
	if now < c.nextPulse {
		return PulseMessage{}, false
	}
	c.nextPulse = now + pulseIntervalSec

	snap := c.tracker.Snapshot()
	msg := PulseMessage{
		Type:      "pulse",
		State:     string(snap.State),
		Beats:     snap.Beats,
		BeatCount: snap.BeatCount,
	}
	if snap.Channel >= 0 && snap.Channel < len(c.pulseNames) {
		name := c.pulseNames[snap.Channel]
		msg.Channel = &name
	}
	if snap.HasBPM {
		v := snap.BPM
		msg.BPM = &v
	}
	if snap.HasRMSSD {
		v := snap.RMSSD
		msg.RMSSD = &v
	}
	if snap.HasIBI {
		v := snap.IBIMs
		msg.IBIMs = &v
	}
	return msg, true
}
```

```js
  // PPG on a classic headband. They are alternatives; no device has both.

  // buildPulse lays out the Pulse panel: one trace for the selected channel,
  // plus a points-only series carrying the detected beats.
  function buildPulse(stream, panel) {
    var c = colors();
    panel.textContent = '';

    var plot = document.createElement('div');
    plot.className = 'plot';
    panel.appendChild(plot);

    var u = new uPlot({
      width: chartWidth(plot),
      height: 168,
      padding: [8, 10, 0, 0],
      legend: { show: false },
      cursor: { y: false, drag: { x: false, y: false, setScale: false } },
      scales: { x: { time: false, auto: false }, y: { auto: true } },
      axes: [xAxis(c, true), yAxis(c, 62)],
      series: [
        {},
        { stroke: seriesColor(c, 0), width: 1.25, points: { show: false } },
        // Beat markers. Drawn as points on a series that is NaN everywhere
        // except at a detected beat, so uPlot spaces them for us instead of
        // needing a custom draw hook.
        {
          stroke: seriesColor(c, 1),
          width: 0,
          points: { show: true, size: 7, fill: seriesColor(c, 1) }
        }
      ]
    }, [new Float64Array(0), new Float64Array(0), new Float64Array(0)], plot);

    stream.pulseChart = u;
  }

  // beatSeries maps beat timestamps onto the trace's x grid, returning a series
  // that is NaN except at the sample nearest each beat.
  function beatSeries(xs, len, ys, beats) {
    var out = new Float64Array(len);
    for (var i = 0; i < len; i++) out[i] = NaN;
    if (!beats || !len) return out;

    var j = 0;
    for (var b = 0; b < beats.length; b++) {
      var t = beats[b];
      if (t < xs[0] || t > xs[len - 1]) continue;
      while (j < len - 1 && xs[j + 1] <= t) j++;
      // Snap to whichever neighbour is closer.
      var k = (j + 1 < len && Math.abs(xs[j + 1] - t) < Math.abs(xs[j] - t)) ? j + 1 : j;
      out[k] = ys[k];
    }
    return out;
  }
```
